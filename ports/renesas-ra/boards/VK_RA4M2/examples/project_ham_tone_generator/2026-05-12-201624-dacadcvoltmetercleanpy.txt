  # DAC-ADC voltmeter с OLED дисплей (SSD1306 128x16).                          
  # Pins: P015=DAC, P001=ADC, P301=SCL, P302=SDA                                
                                                                                
  from machine import DAC, ADC, Pin, SoftI2C                                    
  import framebuf                                                               
  import time                                                                   
  import sys                                                                    
  import select                                                                 
                                                                                
  # --- Конфигурация ---                                                        
  VREF = 3.286                                                                  
  BITS = 12                                                                     
  FULL = 1 << BITS                                                              
                                                                                
  PIN_DAC = "P015"                                                              
  PIN_ADC = "P001"                                                              
  PIN_SCL = "P301"                                                              
  PIN_SDA = "P302"                                                              
                                                                                
  OLED_W    = 128                                                               
  OLED_H    = 16                                                                
  OLED_ADDR = 0x3C                                                              
  OLED_FREQ = 400000                                                            
                                                                                
  ADC_AVG_N = 100                                                               
                                                                                
                                                                                
  # --- SSD1306 128x16 ---                                                      
  class SSD1306:                                                                
      _INIT_CMDS = (                                                            
          0xAE, 0xD5, 0x80, 0xA8, 0x0F, 0xD3, 0x00, 0x40,                       
          0x8D, 0x14, 0x20, 0x00, 0xA1, 0xC8, 0xDA, 0x02,                       
          0x81, 0x7F, 0xD9, 0xF1, 0xDB, 0x40, 0xA4, 0xA6, 0xAF,                 
      )                                                                         
                                                                                
      def __init__(self, i2c, addr=OLED_ADDR):                                  
          self.i2c = i2c                                                        
          self.addr = addr                                                      
          self._tx_buf = bytearray(1 + OLED_W * OLED_H // 8)                    
          self._tx_buf[0] = 0x40                                                
          self._fb_view = memoryview(self._tx_buf)[1:]                          
          self.fb = framebuf.FrameBuffer(self._fb_view, OLED_W, OLED_H,         
                                         framebuf.MONO_VLSB)                    
          self._cmd1 = bytearray(2)                                             
          self._cmd1[0] = 0x00                                                  
          self._show_cmd = bytearray(b"\x00\x21\x00\x7F\x22\x00\x01")           
          self._init_panel()                                                    
          self.fb.fill(0)                                                       
          self.show()                                                           
                                                                                
      def _cmd(self, c):                                                        
          self._cmd1[1] = c                                                     
          self.i2c.writeto(self.addr, self._cmd1)                               
                                                                                
      def _init_panel(self):                                                    
          for c in self._INIT_CMDS:                                             
              self._cmd(c)                                                      
                                                                                
      def fill(self, color):                                                    
          self.fb.fill(color)                                                   
                                                                                
      def text(self, s, x, y, color=1):                                         
          self.fb.text(s, x, y, color)                                          
                                                                                
      def show(self):                                                           
          self.i2c.writeto(self.addr, self._show_cmd)                           
          self.i2c.writeto(self.addr, self._tx_buf)                             
                                                                                
                                                                                
  def oled_probe(i2c):                                                          
      devs = i2c.scan()                                                         
      print("I2C scan:", devs)                                                  
      if 0x3C in devs:                                                          
          return 0x3C                                                           
      if 0x3D in devs:                                                          
          return 0x3D                                                           
      return None                                                               
                                                                                
                                                                                
  # --- Hardware init ---                                                       
  dac = DAC(Pin(PIN_DAC))                                                       
  adc = ADC(Pin(PIN_ADC), bits=12)                                              
                                                                                
  dac_value = 2048                                                              
  dac.write(dac_value)                                                          
                                                                                
  i2c = SoftI2C(scl=Pin(PIN_SCL, Pin.OPEN_DRAIN),                               
                sda=Pin(PIN_SDA, Pin.OPEN_DRAIN), freq=OLED_FREQ)               
  addr = oled_probe(i2c)                                                        
  oled = None                                                                   
  if addr is not None:                                                          
      try:                                                                      
          oled = SSD1306(i2c, addr)                                             
          print("OLED ready at addr", hex(addr))                                
      except Exception as e:                                                    
          print("OLED init failed:", e)                                         
          oled = None                                                           
  else:                                                                         
      print("OLED NOT found - headless")                                        
                                                                                
  print("\nDAC-ADC Voltmeter (live + OLED)")                                    
  print("Type DAC value 0..4095 + Enter to set DAC.\n")                         
                                                                                
  # --- Non-blocking REPL input ---                                             
  _poll = select.poll()                                                         
  _poll.register(sys.stdin, select.POLLIN)                                      
  _in_buf = ""                                                                  
                                                                                
                                                                                
  def poll_repl():                                                              
      """Чете chars от stdin без блокиране. Връща нова DAC стойност или         
  None."""                                                                      
      global _in_buf                                                            
      new_val = None                                                            
      while _poll.poll(0):                                                      
          ch = sys.stdin.read(1)                                                
          if ch == "\n" or ch == "\r":                                          
              line = _in_buf.strip()                                            
              _in_buf = ""                                                      
              if line:                                                          
                  try:                                                          
                      v = int(line)                                             
                  except ValueError:                                            
                      print("\n[ERR] not int: {!r}".format(line))               
                      continue                                                  
                  if 0 <= v <= 4095:                                            
                      new_val = v                                               
                  else:                                                         
                      print("\n[ERR] out of range 0..4095: {}".format(v))       
          elif ch == "\x03":                    # Ctrl-C → exit                 
              raise KeyboardInterrupt                                           
          elif ch == "\x08" or ch == "\x7f":    # backspace                     
              _in_buf = _in_buf[:-1]                                            
          else:                                                                 
              _in_buf += ch                                                     
      return new_val                                                            
                                                                                
                                                                                
  def render(dac_v, dac_volts, adc_v, adc_volts, err):                          
      if oled is None:                                                          
          return                                                                
      oled.fill(0)                                                              
      # 128 px / 8 px = 16 chars per row. 8x8 font: 2 rows (y=0, y=8).          
      oled.text("DAC:{:4d} {:5.3f}V".format(dac_v, dac_volts), 0, 0, 1)         
      oled.text("ADC:{:4d} {:+4d}LSB".format(adc_v, err),      0, 8, 1)         
      oled.show()                                                               
                                                                                
                                                                                
  TICK_MS = 100                                # фиксиран период за OLED +      
  REPL.                                                                         
  next_tick = time.ticks_add(time.ticks_ms(), TICK_MS)                          
                                                                                
  while True:                                                                   
      # REPL команда за нов DAC value (всеки tick).                             
      nv = poll_repl()                                                          
      if nv is not None:                                                        
          dac_value = nv                                                        
          dac.write(dac_value)                                                  
          print("\n[DAC <- {}]".format(dac_value))                              
                                                                                
      # ADC усредняване.                                                        
      total = 0                                                                 
      for _ in range(ADC_AVG_N):                                                
          total += adc.read()                                                   
      raw = total // ADC_AVG_N                                                  
                                                                                
      volts_adc = raw * VREF / FULL                                             
      volts_dac = dac_value * VREF / FULL                                       
      err = raw - dac_value                                                     
                                                                                
      # Серийно (single-line overwrite).                                        
      print(                                                                    
          "\rDAC:{:4d} ({:.3f}V) | ADC:{:4d} ({:.3f}V) | ERR:{:+4d} LSB   "     
          .format(dac_value, volts_dac, raw, volts_adc, err),                   
          end=""                                                                
      )                                                                         
                                                                                
      # OLED.                                                                   
      render(dac_value, volts_dac, raw, volts_adc, err)                         
                                                                                
      # Deadline scheduling -> точен 100 ms cadence.                            
      remaining = time.ticks_diff(next_tick, time.ticks_ms())                   
      if remaining > 0:                                                         
          time.sleep_ms(remaining)                                              
      next_tick = time.ticks_add(next_tick, TICK_MS)

  2. SoftI2C scan на P301/P302, ако намери 0x3C → init SSD1306 128×16.
  3. select.poll() регистрира stdin за non-blocking REPL вход.
  4. Loop tick (на всеки 100 ms, deadline-based):
  5. → poll_repl() чете натрупани chars; на Enter парсва int 0..4095 →
  dac.write(nv).
  6. → 100-сампълно ADC усредняване → raw, volts_adc, volts_dac, err.
  7. → REPL: \r-overwrite DAC:#### (#.###V) | ADC:#### (#.###V) | ERR:±### LSB.
  8. → LCD: text2x рендер на {:5.3f}V (ADC volt) на 2× scale, центрирано.
  9. → sleep_ms(remaining) до следващия 100 ms deadline; next_tick += 100.
  10. Ctrl-C прекъсва loop-а; backspace редактира входния буфер.
